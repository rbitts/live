cmake_minimum_required(VERSION 3.22.1)
project(live_offline LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set_property(GLOBAL PROPERTY JOB_POOLS live_compile=6 live_link=2)
set(CMAKE_JOB_POOL_COMPILE live_compile)
set(CMAKE_JOB_POOL_LINK live_link)
set(CMAKE_CXX_FLAGS_DEBUG "-O2 -g")
set(CMAKE_C_FLAGS_DEBUG "-O2 -g")
set(MNN_BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(MNN_LOW_MEMORY ON CACHE BOOL "" FORCE)
set(MNN_BUILD_LLM ON CACHE BOOL "" FORCE)
set(MNN_BUILD_LLM_OMNI ON CACHE BOOL "" FORCE)
set(MNN_SUPPORT_TRANSFORMER_FUSE ON CACHE BOOL "" FORCE)
set(MNN_BUILD_OPENCV ON CACHE BOOL "" FORCE)
set(MNN_IMGCODECS ON CACHE BOOL "" FORCE)
set(MNN_BUILD_AUDIO ON CACHE BOOL "" FORCE)
set(MNN_BUILD_CONVERTER OFF CACHE BOOL "" FORCE)
set(MNN_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(MNN_OPENCL ON CACHE BOOL "" FORCE)
set(MNN_VULKAN OFF CACHE BOOL "" FORCE)
add_subdirectory(../.upstream/mnn mnn EXCLUDE_FROM_ALL)
# Pinned WebRTC speech/noise classifier, independent of the model worker.
set(VAD_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../.upstream/webrtcvad/cbits)
file(GLOB VAD_SOURCES CONFIGURE_DEPENDS
    ${VAD_ROOT}/webrtc/common_audio/signal_processing/*.c
    ${VAD_ROOT}/webrtc/common_audio/third_party/*.c
    ${VAD_ROOT}/webrtc/common_audio/vad/*.c)
add_library(live_vad STATIC ${VAD_SOURCES} ${VAD_ROOT}/webrtc/rtc_base/checks.cc)
set_target_properties(live_vad PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_definitions(live_vad PRIVATE WEBRTC_POSIX)
target_include_directories(live_vad PUBLIC ${VAD_ROOT})
add_library(live_jni SHARED android_jni.cpp android_profile.cpp android_vad.cpp)
target_include_directories(live_jni PRIVATE ../.upstream/mnn/include ../.upstream/mnn/transformers/llm/engine/include)
target_include_directories(live_jni PRIVATE ../.upstream/mnn/transformers/llm/engine/src ../.upstream/mnn/3rd_party)
target_include_directories(live_jni PRIVATE ../.upstream/mnn/3rd_party/OpenCLHeaders)
target_link_libraries(live_jni PRIVATE llm MNN log dl live_vad)
